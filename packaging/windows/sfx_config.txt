;!@Install@!UTF-8!
Title="Aurora Browser"
; Extract to a stable per-user location instead of a temp folder, so the payload
; is unpacked once instead of on every launch, and so the per-version CEF
; directory (%LOCALAPPDATA%\Aurora\cef\<build_id>) sits on the same volume as
; the files it hard-links from.
InstallPath="%LOCALAPPDATA%\\Aurora\\app"
; 2 = overwrite only files older than the ones in the archive (so an update
;     actually replaces libcef.dll and the .pak files), 8 = skip files the
;     system has locked (a previous Aurora that is still shutting down must not
;     turn an update into a failed extraction).
OverwriteMode="2+8"
; The bootstrap launcher, not the browser: it checks the build ID, materializes
; %LOCALAPPDATA%\Aurora\cef\<build_id>\ if needed, and then starts
; aurora_browser.exe. See src/bootstrap/main.cpp.
RunProgram="aurora_bootstrap.exe"
; Fully silent: no extraction dialog, the browser just opens.
GUIMode="2"
;!@InstallEnd@!
